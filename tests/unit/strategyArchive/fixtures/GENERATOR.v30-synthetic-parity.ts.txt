function mulberry32(a: number) { return () => { a |= 0; a = (a + 0x6D2B79F5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
const H = 3_600_000;
function synth(seed: number, n: number, t0: number) {
  const rnd = mulberry32(seed);
  const h1: Candle[] = []; let px = 100;
  // running 4h-ish extremes tracked to aim sweeps at them
  let recentHi = 102, recentLo = 98;
  for (let i = 0; i < n; i++) {
    const center = 100 + 8 * Math.sin(i / 250) + 3 * Math.sin(i / 61);
    const o = px;
    let c = o + (center - o) * 0.06 + (rnd() - 0.5) * 1.4;
    const event = rnd();
    let hi = Math.max(o, c) + rnd() * 0.3;
    let lo = Math.min(o, c) - rnd() * 0.3;
    let vol = 1000 * (0.7 + rnd() * 0.6);
    if (event < 0.05) {           // upside sweep of recent high & reject
      hi = Math.max(hi, recentHi + 0.05 + rnd() * 0.8); c = o - (0.4 + rnd() * 1.4); lo = Math.min(lo, c - rnd() * 0.15); vol *= 2 + rnd() * 2;
    } else if (event < 0.10) {    // downside sweep & reclaim
      lo = Math.min(lo, recentLo - 0.05 - rnd() * 0.8); c = o + (0.4 + rnd() * 1.4); hi = Math.max(hi, c + rnd() * 0.15); vol *= 2 + rnd() * 2;
    } else if (event < 0.13) {    // impulsive bar, thin wick, big vol (may or may not be near level)
      c = o + (rnd() < 0.5 ? -1 : 1) * (1 + rnd() * 2); vol *= 1.5 + rnd();
      hi = Math.max(o, c) + rnd() * 0.1; lo = Math.min(o, c) - rnd() * 0.1;
    }
    c = Math.max(1, c); hi = Math.max(hi, o, c); lo = Math.min(lo, o, c);
    const openTime = t0 + i * H;
    h1.push({ openTime, open: o, high: hi, low: lo, close: c, volume: vol, closeTime: openTime + H - 1, quoteVolume: 0, trades: 0, isClosed: true });
    px = c;
    if (i % 24 === 0) { const w = h1.slice(Math.max(0, i - 48), i + 1); recentHi = Math.max(...w.map(x => x.high)); recentLo = Math.min(...w.map(x => x.low)); }
  }
  const h4: Candle[] = [];
  for (let i = 0; i + 4 <= n; i += 4) {
    const w = h1.slice(i, i + 4); const openTime = w[0]!.openTime;
    h4.push({ openTime, open: w[0]!.open, high: Math.max(...w.map(x => x.high)), low: Math.min(...w.map(x => x.low)), close: w[3]!.close, volume: w.reduce((s, x) => s + x.volume, 0), closeTime: openTime + 4 * H - 1, quoteVolume: 0, trades: 0, isClosed: true });
  }
  return { h1, h4 };
}

const trades: any[] = [];

}
